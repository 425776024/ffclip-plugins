cmake_minimum_required(VERSION 3.24)
project(VideoCutTextWasm LANGUAGES CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Use Emscripten: emcmake cmake -S . -B build")
endif()
if(NOT EXISTS "${SKIA_ROOT}/include/core/SkCanvas.h" OR NOT EXISTS "${SKIA_BUILD}/libskia.a")
  message(FATAL_ERROR "SKIA_ROOT and SKIA_BUILD must point to the pinned WASM Skia build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-O3 -fno-rtti -fexceptions -ffp-contract=off -msimd128)
add_compile_definitions(SK_TRIVIAL_ABI=[[clang::trivial_abi]] SK_FORCE_8_BYTE_ALIGNMENT SKVX_DISABLE_SIMD SK_DISABLE_TRACING)
set(sdk "${CMAKE_CURRENT_SOURCE_DIR}/vendor/videocut/sdk")
add_library(VideoCutBase STATIC "${sdk}/base/src/Sha256.cpp")
target_include_directories(VideoCutBase PUBLIC "${sdk}/base/include")
add_library(nlohmann_json INTERFACE)
add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
target_include_directories(nlohmann_json INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
add_subdirectory("${sdk}/frame" frame)
add_subdirectory("${sdk}/text" text)
add_subdirectory("${sdk}/vector" vector)
add_subdirectory("${sdk}/text_composition" text_composition)
# Only the renderer-neutral packed-alpha helper is needed. No FFmpeg, desktop
# editor, native GPU bridge, filesystem services, Flutter, or Objective-C.
add_library(VideoCutMedia STATIC "${sdk}/media/src/PackedAlphaYuvMerge.cpp")
target_include_directories(VideoCutMedia PUBLIC "${sdk}/media/include")
target_link_libraries(VideoCutMedia PUBLIC VideoCutFrame)
add_library(VideoCutGpuFrameBridge INTERFACE)
add_library(VideoCutGpuExecution INTERFACE)
add_library(skia_all INTERFACE)
add_library(videocut_skia::all ALIAS skia_all)
target_include_directories(skia_all INTERFACE "${SKIA_ROOT}")
file(GLOB skia_archives CONFIGURE_DEPENDS "${SKIA_BUILD}/*.a")
target_link_libraries(skia_all INTERFACE "-Wl,--start-group" ${skia_archives} "-Wl,--end-group")
set(VIDEOCUT_SKIA_REVISION "933b272d039add454e6491efafd2db84a18a92dd")
set(VIDEOCUT_SKIA_ARTIFACT_PROFILE "videocut-skia-wasm-raster-v1")
add_subdirectory("${sdk}/skia_runtime" skia_runtime)
add_executable(videocut-text src/bridge.cpp src/package_decoder.cpp src/composition_plan.cpp)
target_include_directories(videocut-text PRIVATE "${sdk}/skia_runtime/src" "${SKIA_ROOT}")
target_link_libraries(videocut-text PRIVATE VideoCutTextComposition VideoCutSkiaRuntime VideoCutBase nlohmann_json)
set_target_properties(videocut-text PROPERTIES SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")
target_link_options(videocut-text PRIVATE
  --no-entry -O3 -msimd128 -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,worker,node
  -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=67108864 -sMAXIMUM_MEMORY=1073741824
  -sSTACK_SIZE=2097152 -sDISABLE_EXCEPTION_CATCHING=0
  "-sEXPORTED_FUNCTIONS=['_malloc','_free','_vct_raster_lanes','_vct_create','_vct_destroy','_vct_register_asset','_vct_load','_vct_render','_vct_pixels','_vct_result','_vct_sdf_mesh']"
  "-sEXPORTED_RUNTIME_METHODS=['cwrap','UTF8ToString','HEAPU8']")
